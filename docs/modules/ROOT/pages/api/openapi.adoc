<script>
  (function () {
    var cssLink = document.querySelector('link[href*="/css/site.css"]');
    var frame = document.getElementById('redoc-frame');
    if (cssLink && frame) {
      var uiRoot = cssLink.getAttribute('href').replace(/\/css\/site\.css.*$/, '');
      frame.src = uiRoot + '/redoc-static.html';
    }
  })();
</script>
